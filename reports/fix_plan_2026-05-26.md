  ```python
          return self._regime_cache[0]

      self._regime_cache = (result, now)
      return result

  def _fetch_regime_live(self) -> bool:
      """المنطق الحالي لـ _regime_check ينتقل هنا"""
      ...
  ```
- **الفائدة:** 24 طلب → 3 طلبات يومياً + حماية من انقطاع الشبكة
- **الوقت المقدّر:** 15-20 دقيقة
